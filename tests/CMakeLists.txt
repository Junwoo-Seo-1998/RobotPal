cmake_minimum_required(VERSION 3.16)
project(RobotPalTransportTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(transport_policy_test transport_policy_test.cpp)
target_include_directories(transport_policy_test PRIVATE ../RobotPal/include)
target_compile_features(transport_policy_test PRIVATE cxx_std_17)
target_link_libraries(transport_policy_test PRIVATE Threads::Threads)
# Assertions are test checks and must also run in Release configurations.
if(MSVC)
    target_compile_options(transport_policy_test PRIVATE /UNDEBUG)
else()
    target_compile_options(transport_policy_test PRIVATE -UNDEBUG)
endif()
add_test(NAME transport_policy COMMAND transport_policy_test)
if(UNIX)
    add_executable(tcp_transport_test tcp_transport_test.cpp
        ../RobotPal/src/Network/TcpNetworkTransport.cpp
        ../RobotPal/src/Network/NetworkTransport.cpp)
    target_include_directories(tcp_transport_test PRIVATE ../RobotPal/include)
    target_compile_features(tcp_transport_test PRIVATE cxx_std_17)
    target_compile_options(tcp_transport_test PRIVATE -UNDEBUG)
    target_link_libraries(tcp_transport_test PRIVATE Threads::Threads)
    add_test(NAME tcp_transport COMMAND tcp_transport_test)
    set_tests_properties(tcp_transport PROPERTIES TIMEOUT 15)
endif()
